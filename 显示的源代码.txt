      const loadComments = async (requestedPage, append = false, automatic = false) => {
        const list = root.querySelector('[data-comment-list]');
        if (!list || loading) return false;
        stopInitialLoader();
        paginationObserver?.disconnect();
        paginationObserver = null;
        loading = true;
        list.setAttribute('aria-busy', 'true');
        setPaginationStatus('正在加载...');
        syncPaginationControls();
        setDeferredPrompt('', false, true);
        try {
          const params = new URLSearchParams({ cid });
          if (requestedPage !== null) params.set('commentPage', String(requestedPage));
          const response = await fetch(`/api/comments?${params}`, {
            credentials: 'same-origin', headers: { Accept: 'application/json' },
          });
          const data = await response.json();
          if (!response.ok) throw new Error(data.error || '评论加载失败');
          renderComments(data, list, append, automatic);
          return true;
        } catch (error) {
          const message = root.querySelector('[data-comment-message]');
          setCommentMessage(message, error instanceof Error ? error.message : '评论加载失败');
          if (automatic) automaticPaused = true;
          if (!commentsLoaded) setDeferredPrompt('评论加载失败，点击重试。', true, false);
          else {
            renderPagination(activePagination);
            setPaginationStatus('加载失败，请重试。');
          }
          return false;
        } finally {
          loading = false;
          list.removeAttribute('aria-busy');
          syncPaginationControls();
        }
      };

      const loadMetadata = (requestedPage = null, includeComments = false) => {
        if (metadataLoaded) return Promise.resolve();
        if (metadataRequest) return metadataRequest;
        metadataRequest = (async () => {
          const params = new URLSearchParams({ cid });
          if (!includeComments) params.set('includeComments', '0');
          const requestedCommentPage = requestedPage === null ? root.dataset.currentPage : String(requestedPage);
          if (requestedCommentPage) params.set('commentPage', requestedCommentPage);
          const response = await fetch(`/api/comments?${params}`, {
            credentials: 'same-origin', headers: { Accept: 'application/json' },
          });
          const data = await response.json();
          if (!response.ok) throw new Error(data.error || '评论配置加载失败');
          renderFrame(root, data);
          applyCooldownLock();
          metadataLoaded = true;
          return data;
        })().finally(() => {
          metadataRequest = null;
        });
        return metadataRequest;

        } catch (error) {
          setCommentMessage(message, error instanceof Error ? error.message : '评论发送失败');
        } finally {

        }
      });

      const bootstrap = async () => {
        stopInitialLoader();
        if (initialMode === 'manual') {

          } catch (error) {

          return;
        }

          dwellTimer = null;
          setDwellLoading(false);
        };
        const startDwellTimer = () => {

          setDwellLoading(true);

            dwellTimer = null;
            if (componentVisible) void bootstrap();
            else setDwellLoading(false);
          }, 3_000);
        };

        if (!('IntersectionObserver' in window)) {
          componentVisible = true;
          startDwellTimer();
          return;
        }
        componentObserver = new IntersectionObserver(entries => {
          componentVisible = entries.some(entry => entry.isIntersecting);
          if (!componentVisible) {
            stopDwellTimer();
            return;
          }
          startDwellTimer();

        componentObserver.observe(deferred);
      };

        if (event.persisted) restoreCooldown();

      window.addEventListener('instantclick:newpage', cleanupStalePage);

          frame.append(anchor, button);

        const copyText = async (text, trigger) => {
          let copied = false;
          try {
            if (navigator.clipboard?.writeText && window.isSecureContext) {
              await navigator.clipboard.writeText(text);
              copied = true;
            }
          } catch {}
          if (!copied) {
            const textarea = document.createElement('textarea');
            textarea.value = text;
            textarea.setAttribute('readonly', '');
            textarea.style.position = 'fixed';
            textarea.style.opacity = '0';
            textarea.style.pointerEvents = 'none';
            const copyHost = trigger?.closest('dialog[open]') || document.body;
            copyHost.append(textarea);
            textarea.focus({ preventScroll: true });
            textarea.select();
            textarea.setSelectionRange(0, textarea.value.length);
            const handleCopy = event => {
              if (!event.clipboardData) return;
              event.clipboardData.setData('text/plain', text);
              event.preventDefault();
              copied = true;
            };
            textarea.addEventListener('copy', handleCopy);
            try { document.execCommand('copy'); } catch {}
            textarea.removeEventListener('copy', handleCopy);
            textarea.remove();
          }
          return copied;

          } else if (pre) {

            clone.querySelector('[data-warm-copy]')?.remove();

          } else if (isInline) {

            if (!code) return;
            text = code.textContent || '';
          } else {
            return;

          const originalTitle = button.getAttribute('title') || originalLabel;

            button.dataset.copied = 'false';
            button.setAttribute('aria-label', originalLabel);
            button.title = originalTitle;
            button.innerHTML = originalHtml;
          }, 1200);
        });

        const navStyle = window.getComputedStyle(headerNav);
        const innerGap = Number.parseFloat(innerStyle.columnGap) || 0;
        const rightGap = 12;
        const navGap = Number.parseFloat(navStyle.columnGap) || 0;
        const navItems = Array.from(headerNav.children);
        const navItemsWidth = navItems.reduce((total, item) => total + item.getBoundingClientRect().width, 0)
          + Math.max(0, navItems.length - 1) * navGap;
        const availableRightWidth = headerInner.clientWidth - headerBrand.getBoundingClientRect().width - innerGap;
        const requiredRightWidth = headerTools.getBoundingClientRect().width + rightGap + navItemsWidth;
        header.classList.toggle('warm-header--tools-inline', requiredRightWidth <= availableRightWidth + 1);
      };
      updateHeaderToolsLayout();
      if ('ResizeObserver' in window && headerInner) {
        headerLayoutObserver = new ResizeObserver(updateHeaderToolsLayout);
        headerLayoutObserver.observe(headerInner);
        if (headerBrand) headerLayoutObserver.observe(headerBrand);
        if (headerNav) headerLayoutObserver.observe(headerNav);
        if (headerTools) headerLayoutObserver.observe(headerTools);
      } else {
        window.addEventListener('resize', updateHeaderToolsLayout, { passive: true, signal: pageAbort.signal });
      }
      compactHeader.addEventListener?.('change', updateHeaderToolsLayout, { signal: pageAbort.signal });
      window.addEventListener('load', updateHeaderToolsLayout, { once: true, signal: pageAbort.signal });

      const hoverCapable = window.matchMedia('(hover: hover) and (pointer: fine)');
      document.querySelectorAll('.warm-nav__dropdown').forEach(dropdown => {
        let openedByHover = false;
        dropdown.addEventListener('mouseenter', () => {
          if (hoverCapable.matches && !dropdown.open) {
            dropdown.open = true;
            openedByHover = true;
          }
        }, { signal: pageAbort.signal });
        dropdown.addEventListener('click', event => {
          if (!event.target.closest('summary')) return;
          // The first pointer click must not undo the preceding hover-open.
          // Keyboard activation (detail=0) keeps the native details toggle.
          if (openedByHover && event.detail > 0) event.preventDefault();
          openedByHover = false;
        }, { signal: pageAbort.signal });
        dropdown.addEventListener('mouseleave', () => {
          openedByHover = false;
          if (hoverCapable.matches && !dropdown.contains(document.activeElement)) dropdown.open = false;
        }, { signal: pageAbort.signal });
        dropdown.addEventListener('keydown', event => {
          if (event.key !== 'Escape') return;
          dropdown.open = false;
          dropdown.querySelector('summary')?.focus();
        }, { signal: pageAbort.signal });
        dropdown.addEventListener('focusout', event => {
          // iOS Safari can report a null relatedTarget while a touch tap moves
          // from <summary> to a menu link. Keep the menu until outside-click handles it.
          if (event.relatedTarget && !dropdown.contains(event.relatedTarget)) dropdown.open = false;
        }, { signal: pageAbort.signal });
      });
    })();
